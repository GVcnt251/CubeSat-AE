cmake_minimum_required(VERSION 3.22)
option(BENCH_CAN_TELEMETRY "Poll the three reaction wheels for read-only bench telemetry" OFF)
option(BENCH_CAN_DIAGNOSTICS "Report CAN controller errors and observed reply headers" OFF)
option(BENCH_USB_CAMERA "Stream camera JPEGs over the USB console" OFF)
if(BENCH_CAN_DIAGNOSTICS AND NOT BENCH_CAN_TELEMETRY)
    message(FATAL_ERROR "BENCH_CAN_DIAGNOSTICS requires BENCH_CAN_TELEMETRY")
endif()
if(BENCH_USB_CAMERA)
    list(APPEND EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/camera_component")
    list(APPEND COMPONENTS main bench_camera)
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
if(NOT BENCH_USB_CAMERA)
    idf_build_set_property(MINIMAL_BUILD ON)
endif()
project(cubo_imu_usb_20hz)
